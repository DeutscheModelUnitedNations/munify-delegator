/*

 * Advisory only: the care team decides every pair (`possibleDuplicate.status`).
 *
 * TODO(privacy): split the personal data off `user` into a `userData` table, as the AVV app does:
 * `user` keeps the id the 13 restricting foreign keys point at, and deleting an account deletes
 * only its `userData` - today an erasure request needs hand-written SQL. Once that exists, keep a
 * tombstone of a deleted account that carried a care note (keyed hashes of phone, email local part,
 * emergency numbers and birthday + name, with an expiry) and match new accounts against it too.
 * Before building it: the legal basis for keeping notes and fingerprints after an erasure request

/** What the matcher reads of an account, and nothing else. */
const PROFILE_COLUMNS = {
	id: true,
	givenName: true,
	familyName: true,
	birthday: true,
	email: true,
	phone: true,
	emergencyContacts: true,
	street: true,
	zip: true,
	country: true
} as const;
 
/**
 * The accounts a scanned one is compared with: who attended a conference that is over, or has a
 * care note. Accounts that never got that far have no history to lose, and leaving them out keeps
 * the personal data the comparison touches to what it is for.
 */
function loadPool() {
	return db.query.user.findMany({
		columns: PROFILE_COLUMNS,
		where: {
			OR: [
				participatesIn({ endConference: { lt: new Date() } }),
				{ globalNotes: { isNotNull: true, ne: '' } }
			]
		}
	});
}
 
/** Which accounts a scan compares: one account, or everyone taking part in a conference. */
type UserFilter = NonNullable<NonNullable<Parameters<typeof db.query.user.findMany>[0]>['where']>;
 
/** Rows per statement: postgres takes at most 65535 parameters in one. */
const BATCH = 5000;
 
function batches<T>(items: T[]) {
	const result: T[][] = [];
	for (let i = 0; i < items.length; i += BATCH) result.push(items.slice(i, i + BATCH));
	return result;
}
 
/**
 * Stores what a scan found: new pairs open, known ones get the fresh score and reasons but keep
 * their decision. An open pair the scan no longer finds is dropped - but only where the scan
 * could have found it, i.e. its other account is in the pool; otherwise it came from scanning that
 * other account, and stays.
 */
async function store({
	scannedWhere,
	scannedIds,
	poolIds,
	pairs
}: {
	scannedWhere: UserFilter;
	scannedIds: Set<string>;
	poolIds: Set<string>;
	pairs: DuplicatePair[];
}) {
	const found = new Set(pairs.map((pair) => `${pair.userId}|${pair.candidateId}`));
 
	const written = await db.transaction(async (tx) => {
		const existing = await tx.query.possibleDuplicate.findMany({
			columns: { id: true, userId: true, candidateId: true },
			where: { status: 'OPEN', OR: [{ user: scannedWhere }, { candidate: scannedWhere }] }
		});
		const stale = existing.filter((row) => {
			if (found.has(`${row.userId}|${row.candidateId}`)) return false;
			const other = scannedIds.has(row.userId) ? row.candidateId : row.userId;
			return poolIds.has(other);
		});
		for (const batch of batches(stale)) {
			await tx.delete(schema.possibleDuplicate).where(
				whereRows({
					table: schema.possibleDuplicate,
					filter: { id: { in: batch.map((row) => row.id) } }
				})
			);
		}
 
		const upserted: { id: string; inserted: boolean }[] = [];
		for (const batch of batches(pairs)) {
			const rows = await tx
				.insert(schema.possibleDuplicate)
				.values(batch)
				.onConflictDoUpdate({
					target: [schema.possibleDuplicate.userId, schema.possibleDuplicate.candidateId],
					set: {
						score: sql`excluded.score`,
						reasons: sql`excluded.reasons`,
						updatedAt: sql`CURRENT_TIMESTAMP`
					}
				})
				// xmax is 0 on a row the statement inserted, set on one it updated
				.returning({ id: schema.possibleDuplicate.id, inserted: sql<boolean>`xmax = 0` });
			upserted.push(...rows);
		}
		return { removed: stale.length > 0, upserted };
	});
	if (written.removed) possibleDuplicatePubsub.removed();
	if (written.upserted.some((row) => row.inserted)) possibleDuplicatePubsub.created();
	const refreshed = written.upserted.filter((row) => !row.inserted).map((row) => row.id);
	if (refreshed.length > 0) possibleDuplicatePubsub.updated(refreshed);
 
	return pairs.length;
}
 
/** How many accounts are indexed or matched before the event loop gets a turn. */
const CHUNK = 2000;
 
/** Runs `work` over `items` a chunk at a time, letting other requests through in between. */
async function inChunks<T>({ items, work }: { items: T[]; work: (chunk: T[]) => void }) {
	for (let i = 0; i < items.length; i += CHUNK) {
		work(items.slice(i, i + CHUNK));
		await new Promise((resolve) => setImmediate(resolve));
	}
}
 
async function scan(scannedWhere: UserFilter) {
	const [scanned, pool] = await Promise.all([
		db.query.user.findMany({ columns: PROFILE_COLUMNS, where: scannedWhere }),
		loadPool()
	]);
	const index = createDuplicateIndex();
	await inChunks<MatchProfile>({ items: pool, work: (chunk) => index.add(chunk) });
	await inChunks<MatchProfile>({ items: scanned, work: (chunk) => index.match(chunk) });
	return store({
		scannedWhere,
		scannedIds: new Set(scanned.map((user) => user.id)),
		poolIds: new Set(pool.map((user) => user.id)),
		pairs: index.found()
	});
}
 
/** Compares everyone taking part in the conference with the pool; answers how many pairs it holds. */
export function scanConferenceForDuplicates(conferenceId: string) {
	return scan(participatesIn({ id: conferenceId }));
}
 
/**
 * Compares one account with the pool, after it was created or its profile changed. Runs in the
 * background of the request that changed it: a failure is logged, never the request's problem.
 */
export function scanUserForDuplicatesInBackground(userId: string) {
	scan({ id: userId }).catch((error: unknown) => {
		logger.error('Scanning an account for possible duplicates failed', { userId, err: error });
	});
}
 
/** A conference an account of a possible duplicate took part in, and as what. */
export type Attendance = {
	conferenceId: string;
	title: string;
	startConference: Date;
	role: 'DELEGATION_MEMBER' | 'SINGLE_PARTICIPANT' | 'SUPERVISOR';
};
 
const CONFERENCE_COLUMNS = { columns: { id: true, title: true, startConference: true } } as const;
 
/**
 * Every conference the account took part in, newest first: what tells the care team whether the
 * other account is the one they remember. Conference titles and dates are public, so this does
 * not go through the registration rows' read rules, which would show far more.
 */
export async function attendancesOf(userId: string): Promise<Attendance[]> {
	const [memberships, singles, supervisions] = await Promise.all([
		db.query.delegationMember.findMany({
			where: { participant: { userId } },
			with: { delegation: { columns: {}, with: { conference: CONFERENCE_COLUMNS } } }
		}),
		db.query.singleParticipant.findMany({
			where: { participant: { userId } },
			with: { conference: CONFERENCE_COLUMNS }
		}),
		db.query.conferenceSupervisor.findMany({
			where: { participant: { userId } },
			with: { conference: CONFERENCE_COLUMNS }
		})
	]);
	const attendance = ({
		conference,
		role
	}: {
		conference: { id: string; title: string; startConference: Date };
		role: Attendance['role'];
	}): Attendance => ({
		conferenceId: conference.id,
		title: conference.title,
		startConference: conference.startConference,
		role
	});
	return [
		...memberships.map((m) =>
			attendance({ conference: m.delegation.conference, role: 'DELEGATION_MEMBER' })
		),
		...singles.map((s) => attendance({ conference: s.conference, role: 'SINGLE_PARTICIPANT' })),
		...supervisions.map((s) => attendance({ conference: s.conference, role: 'SUPERVISOR' }))
	].sort((a, b) => b.startConference.getTime() - a.startConference.getTime());
}
 
/** Everyone `accounts` reach through confirmed pairs, as they were linked so far. */
async function linkedAccounts({ tx, accounts }: { tx: Transaction; accounts: string[] }) {
	const reached = new Set(accounts);
	let frontier = [...reached];
	while (frontier.length > 0) {
		const linked = await tx.query.possibleDuplicate.findMany({
			columns: { userId: true, candidateId: true },
			where: {
				status: 'CONFIRMED',
				OR: [{ userId: { in: frontier } }, { candidateId: { in: frontier } }]
			}
		});
		frontier = linked
			.flatMap((row) => [row.userId, row.candidateId])
			.filter((account) => !reached.has(account));
		for (const account of frontier) reached.add(account);
	}
	return [...reached];
}
 
async function assertNoLinkConflict({
	tx,
	conflict
}: {
	tx: Transaction;
	conflict: LinkPair | undefined;
}) {
	if (!conflict) return;
	const names = await tx.query.user.findMany({
		columns: { id: true, givenName: true, familyName: true },
		where: { id: { in: [conflict.userId, conflict.candidateId] } }
	});
	const [first, second] = names.map((user) => `${user.givenName} ${user.familyName}`);
	throw apiError({
		code: 'CONFLICT',
		message: m.possibleDuplicateLinkConflict({ first: first ?? '', second: second ?? '' })
	});
}
 
async function writeLinkChanges({
	tx,
	changes,
	decidedById,
	decidedAt
}: {
	tx: Transaction;
	changes: LinkChanges;
	decidedById: string;
	decidedAt: Date | null;
}) {
	const pairsIn = (ids: string[]) =>
		whereRows({ table: schema.possibleDuplicate, filter: { id: { in: ids } } });
	const linked = { status: 'CONFIRMED' as const, inferred: true, decidedById, decidedAt };
	if (changes.confirm.length > 0) {
		await tx.update(schema.possibleDuplicate).set(linked).where(pairsIn(changes.confirm));
	}
	if (changes.reopen.length > 0) {
		await tx
			.update(schema.possibleDuplicate)
			.set({ status: 'OPEN', inferred: false, decidedById: null, decidedAt: null })
			.where(pairsIn(changes.reopen));
	}
	if (changes.remove.length > 0) {
		await tx.delete(schema.possibleDuplicate).where(pairsIn(changes.remove));
	}
	if (changes.insert.length > 0) {
		await tx
			.insert(schema.possibleDuplicate)
			.values(changes.insert.map((link) => ({ ...link, score: 0, reasons: [], ...linked })));
	}
}
 
/**
 * Settles a pair: `CONFIRMED` (one person), `DISMISSED` (two people), or back to `OPEN`, which
 * forgets who decided and when. The links follow (`reconcileLinks`): confirming pairs every account
 * of the person with every other one, reopening or dismissing takes back what only this pair
 * implied. CONFLICT when a confirmation would make one person of two accounts a pair says are two.
 */
export async function decidePossibleDuplicate({
	id,
	status,
	decidedById
}: {
	id: string;
	status: (typeof schema.possibleDuplicate.$inferSelect)['status'];
	decidedById: string;
}) {
	const decided = status !== 'OPEN';
	const decidedAt = decided ? new Date() : null;
	const touched = await db.transaction(async (tx) => {
		// decisions are rare; one at a time keeps two of them from linking past each other
		await tx.execute(sql`select pg_advisory_xact_lock(hashtext('possible_duplicate_links'))`);
		const [pair] = await tx
			.update(schema.possibleDuplicate)
			.set({ status, inferred: false, decidedById: decided ? decidedById : null, decidedAt })
			.where(whereRows({ table: schema.possibleDuplicate, filter: { id } }))
			.returning({
				userId: schema.possibleDuplicate.userId,
				candidateId: schema.possibleDuplicate.candidateId
			});
		if (!pair) return { pairIds: [id], userIds: [], created: false, removed: false };
 
		const among = await linkedAccounts({ tx, accounts: [pair.userId, pair.candidateId] });
		const pairs = await tx.query.possibleDuplicate.findMany({
			columns: {
				id: true,
				userId: true,
				candidateId: true,
				status: true,
				inferred: true,
				reasons: true
			},
			where: { userId: { in: among }, candidateId: { in: among } }
		});
		const changes = reconcileLinks(pairs);
		await assertNoLinkConflict({ tx, conflict: changes.conflicts[0] });
		await writeLinkChanges({ tx, changes, decidedById, decidedAt });
		return {
			pairIds: [id, ...changes.confirm, ...changes.reopen],
			// what of each account the care team may read follows the decision
			userIds: among,
			created: changes.insert.length > 0,
			removed: changes.remove.length > 0
		};
	});
 
	possibleDuplicatePubsub.updated(touched.pairIds);
	if (touched.created) possibleDuplicatePubsub.created();
	if (touched.removed) possibleDuplicatePubsub.removed();
	if (touched.userIds.length > 0) userPubsub.updated(touched.userIds);
}
