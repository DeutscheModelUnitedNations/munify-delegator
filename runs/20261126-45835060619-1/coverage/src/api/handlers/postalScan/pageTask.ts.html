/** A user's part in a conference. */
async function participantIdOf(who: { userId: string; conferenceId: string }) {
	const participant = await db.query.conferenceParticipant.findFirst({
		where: who,
		columns: { id: true }
	});
	return participant?.id;
}
 
/** Keeps a page, as the file it was cut out as or that file turned upright, and takes in what it says. */
async function keepPage({
	args: { participant: participantId, page, pageKey, file, turned },
	task: { userId, fileId }
}: {
	args: KeepArgs<string>;
	task: ScanTask;
}): Promise<KeptPage> {
	const outcome = await takeInPage({
		participantId,
		source: 'WEBDAV',
		page: { pageKey, fields: page.fields },
		readings: page.readings,
		uploadedById: userId,
		storeFile: async () =>
			turned
				? storeGeneratedFile({
						userId,
						purpose: filePurpose({ kind: 'postalScan', field: 'scan' }),
						fileName: file.name,
						mimeType: file.type,
						bytes: file.bytes
					})
				: fileId
	});
	// it answers `null` only without a file
	return { outcome: outcome ?? 'known' };
}
