import { db } from '#api/db/db';
import { abilityBuilder, type ApiContext } from '#api/rumble';
import {
	hop,
	idIn,
	isInOwnDelegation,
	isTeamMemberOfConference,
	ledDelegationIds,
	PARTICIPANT_CARE_ROLES,
	supervisedDelegationIds,
	systemAdmin
} from '#api/services/authHelper/rules';
import { liveWhere } from '#api/services/authHelper/callerFacts';
import { assertFindFirstExists } from '@m1212e/rumble';
import { m } from '#lib/paraglide/messages';
import { apiError } from '#api/services/errors';
 
export function registerRoleApplicationAbilities() {
	abilityBuilder.roleApplication.allow(['read', 'update', 'delete']).when(systemAdmin);
 
	// Every delegate reads their own delegation's role applications.
	abilityBuilder.roleApplication.allow('read').when(liveWhere(isInOwnDelegation));
 
	// The head delegate may update and delete their delegation's role applications until it has
	// applied - the same line the delegation's own update ability draws.
	abilityBuilder.roleApplication.allow(['update', 'delete']).when(
		liveWhere((args) => {
			const led = ledDelegationIds(args, { notApplied: true });
			return led && { delegationId: idIn(led) };
		})
	);
 
	// Participant care (project management and participant care) reads, updates and deletes the role
	// applications of its conference, also after the delegation applied, as it may the delegation.
	abilityBuilder.roleApplication.allow(['read', 'update', 'delete']).when(
		liveWhere((args) => {
			const delegation = isTeamMemberOfConference({ ...args, roles: PARTICIPANT_CARE_ROLES });
			return hop(delegation, (delegation) => ({ delegation }));
		})
	);
 
	// Supervisors read those of the delegations they supervise.
	abilityBuilder.roleApplication.allow('read').when(
		liveWhere((args) => {
			const ids = supervisedDelegationIds(args);
			return ids && { delegationId: idIn(ids) };
		})
	);
}
 
/**
 * Refuses (NOT_FOUND) unless the caller's update ability reaches the delegation - its head delegate
 * until it has applied, or `manageParticipants` (project management and participant care) in its
 * conference. The role being a nation or non-state actor of its conference is the database's check
 * (`role_application_same_conference`, `role_application_role_kind`).
 */
export async function assertCanCreateRoleApplication({
	ctx,
	args
}: {
	ctx: ApiContext;
	args: { delegationId: string; roleId: string };
}) {
	await db.query.delegation
		.findFirst({
			...(await ctx.abilities.delegation.filter('update')).merge({
				where: { id: args.delegationId }
			}).query.single,
			columns: { id: true }
		})
		.then(assertFindFirstExists);
}
 
/**
 * Refuses (NOT_FOUND) unless the caller's delete ability reaches the role application: the head
 * delegate until the delegation has applied, or `manageParticipants` (project management and
 * participant care) in its conference. Returns its id and delegation, whose ranks close the gap
 * afterwards.
 */
export async function assertCanDeleteRoleApplication({ ctx, id }: { ctx: ApiContext; id: string }) {
	const deletable = await db.query.roleApplication.findFirst({
		...(await ctx.abilities.roleApplication.filter('delete')).merge({ where: { id } }).query.single,
		columns: { id: true, delegationId: true }
	});
	if (!deletable) {
		throw apiError({ code: 'NOT_FOUND', message: m.deleteNotFoundOrForbidden() });
	}
	return deletable;
}
 
/**
 * Refuses (NOT_FOUND) unless the caller's update ability reaches both role applications (as for
 * deleting one), and (INVALID_INPUT) unless they belong to one delegation. Returns them.
 */
export async function assertCanSwapRoleApplicationRanks({
	ctx,
	firstId,
	secondId
}: {
	ctx: ApiContext;
	firstId: string;
	secondId: string;
}) {
	const updatable = await ctx.abilities.roleApplication.filter('update');
	const [first, second] = await Promise.all([
		db.query.roleApplication
			.findFirst(updatable.merge({ where: { id: firstId } }).query.single)
			.then(assertFindFirstExists),
		db.query.roleApplication
			.findFirst(updatable.merge({ where: { id: secondId } }).query.single)
			.then(assertFindFirstExists)
	]);
	if (first.delegationId !== second.delegationId) {
		throw apiError({ code: 'INVALID_INPUT', message: m.roleApplicationsOfDifferentDelegations() });
	}
	return { first, second };
}
